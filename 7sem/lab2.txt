ena s class
ser pass

pass cisco
login

pass cisco
login
logg s

sh

no sh

ip addr 192.168.1.1 255.255.255.0
no sh
exit

aaa new-model
aaa authentication login default local
aaa authorization exec default local
aaa authorization console

username root privilege 15 view root secret root

parser view default_user
secret user
commands exec exclude show ip route
commands exec exclude show banner
commands exec exclude show parser
commands exec include all show ip 
commands exec include ping
exit

username user view default_user secret user

snmp-server view all iso included
snmp-server group readers v3 priv read all

snmp-server user readers_admin readers v3 auth sha AuthPasswd priv aes 128 PrivPasswd

snmp mib nhrp
snmp-server enable traps
snmp-server group notifiers v3 priv

snmp-server user notifiers_admin notifiers v3 auth sha AuthPasswd priv aes 128 PrivPasswd

snmp-server host 192.168.1.10 version 3 priv notifiers_admin

end
